  <script>
    function getNumber(id) {
      var v = parseFloat(document.getElementById(id).value);
      return isNaN(v) ? 0 : v;
    }

    // Old regime tax slabs (FY 2024–25), with 87A rebate and 4% cess
    function calculateOldRegimeTax(taxable) {
      let tax = 0;

      if (taxable <= 250000) {
        tax = 0;
      } else if (taxable <= 500000) {
        tax = (taxable - 250000) * 0.05;
      } else if (taxable <= 1000000) {
        tax = 250000 * 0.05 + (taxable - 500000) * 0.20;
      } else {
        tax = 250000 * 0.05 + 500000 * 0.20 + (taxable - 1000000) * 0.30;
      }

      // Rebate u/s 87A if taxable income <= 5L (resident)
      if (taxable <= 500000) {
        tax = 0;
      }

      const cess = tax * 0.04;
      return {
        basicTax: Math.round(tax),
        cess: Math.round(cess),
        totalTax: Math.round(tax + cess)
      };
    }

    // New regime tax slabs (FY 2024–25), with 87A rebate and 4% cess
    function calculateNewRegimeTax(taxable) {
      let tax = 0;

      if (taxable <= 300000) {
        tax = 0;
      } else if (taxable <= 600000) {
        tax = (taxable - 300000) * 0.05;
      } else if (taxable <= 900000) {
        tax = 300000 * 0.05 + (taxable - 600000) * 0.10;
      } else if (taxable <= 1200000) {
        tax = 300000 * 0.05 + 300000 * 0.10 + (taxable - 900000) * 0.15;
      } else if (taxable <= 1500000) {
        tax = 300000 * 0.05 + 300000 * 0.10 + 300000 * 0.15 + (taxable - 1200000) * 0.20;
      } else {
        tax = 300000 * 0.05 + 300000 * 0.10 + 300000 * 0.15 + 300000 * 0.20 + (taxable - 1500000) * 0.30;
      }

      // Rebate u/s 87A (new regime) if taxable income <= 7L
      if (taxable <= 700000) {
        tax = 0;
      }

      const cess = tax * 0.04;
      return {
        basicTax: Math.round(tax),
        cess: Math.round(cess),
        totalTax: Math.round(tax + cess)
      };
    }

    function runCalculator() {
      const errorDiv = document.getElementById('error');
      const resultsDiv = document.getElementById('results');
      errorDiv.textContent = '';
      resultsDiv.innerHTML = '';

      const age = getNumber('age');
      const isSalaried = document.getElementById('isSalaried').value === 'yes';

      if (!age || age < 18) {
        errorDiv.textContent = 'Please enter a valid age (18 or above).';
        return;
      }

      const salary   = getNumber('salaryIncome');
      const business = getNumber('businessIncome');
      const rent     = getNumber('rentIncome');
      const other    = getNumber('otherIncome');
      const capital  = getNumber('capitalGains');

      const grossIncome = salary + business + rent + other + capital;

      if (grossIncome <= 0) {
        errorDiv.textContent = 'Please enter at least one positive income amount.';
        return;
      }

      // Standard deduction for salaried/pensioner (simplified)
      const standardDeduction = isSalaried ? 50000 : 0;
      const incomeAfterStandard = Math.max(0, grossIncome - standardDeduction);

      // Deductions for old regime
      let d80c          = getNumber('d80c');
      let d80d          = getNumber('d80d');
      let d80ccd1b      = getNumber('d80ccd1b');
      let d80e          = getNumber('d80e');
      let d80g          = getNumber('d80g');
      let homeLoanInt   = getNumber('homeLoanInterest');

      const max80C            = 150000;
      const max80CCD1B        = 50000;
      const maxHomeLoanInt    = 200000;
      const max80DCombined    = 75000; // simplified combined cap (self+family+parents)

      const used80C           = Math.min(d80c, max80C);
      const used80D           = Math.min(d80d, max80DCombined);
      const used80CCD1B       = Math.min(d80ccd1b, max80CCD1B);
      const usedHomeLoanInt   = Math.min(homeLoanInt, maxHomeLoanInt);

      const totalDeductionsOld = used80C + used80D + used80CCD1B + d80e + d80g + usedHomeLoanInt;

      const taxableOld = Math.max(0, incomeAfterStandard - totalDeductionsOld);
      const taxableNew = Math.max(0, incomeAfterStandard); // most deductions not allowed in new regime

      const oldResult = calculateOldRegimeTax(taxableOld);
      const newResult = calculateNewRegimeTax(taxableNew);

      const betterRegime =
        oldResult.totalTax < newResult.totalTax ? 'Old' :
        newResult.totalTax < oldResult.totalTax ? 'New' :
        'Both similar';

      // For tax-saving hints
      const remaining80C      = Math.max(0, max80C - used80C);
      const remaining80CCD1B  = Math.max(0, max80CCD1B - used80CCD1B);

      // Approx marginal rate in old regime (for suggestion only)
      let marginalRateOld = 0;
      if (taxableOld > 1000000)      marginalRateOld = 0.30;
      else if (taxableOld > 500000)  marginalRateOld = 0.20;
      else if (taxableOld > 250000)  marginalRateOld = 0.05;
      else                           marginalRateOld = 0;

      let potential80CSaving = Math.round(remaining80C * marginalRateOld);
      let potentialNpsSaving = Math.round(remaining80CCD1B * marginalRateOld);

      let html = '';

      html += '<div class="result-box">';
      html += '<h3>Tax summary</h3>';
      html += '<p><strong>Gross total income:</strong> ₹' + grossIncome.toLocaleString('en-IN') + '</p>';
      html += '<p><strong>Standard deduction:</strong> ₹' + standardDeduction.toLocaleString('en-IN') + '</p>';
      html += '<p><strong>Income after standard deduction:</strong> ₹' + incomeAfterStandard.toLocaleString('en-IN') + '</p>';

      html += '<h4>Old tax regime</h4>';
      html += '<p>Eligible deductions (approx): ₹' + totalDeductionsOld.toLocaleString('en-IN') + '</p>';
      html += '<p>Taxable income (Old regime): ₹' + taxableOld.toLocaleString('en-IN') + '</p>';
      html += '<p>Income tax (before cess): ₹' + oldResult.basicTax.toLocaleString('en-IN') + '</p>';
      html += '<p>Health & education cess (4%): ₹' + oldResult.cess.toLocaleString('en-IN') + '</p>';
      html += '<p><strong>Total tax – Old regime:</strong> ₹' + oldResult.totalTax.toLocaleString('en-IN') + '</p>';

      html += '<h4>New tax regime</h4>';
      html += '<p>Taxable income (New regime): ₹' + taxableNew.toLocaleString('en-IN') + '</p>';
      html += '<p>Income tax (before cess): ₹' + newResult.basicTax.toLocaleString('en-IN') + '</p>';
      html += '<p>Health & education cess (4%): ₹' + newResult.cess.toLocaleString('en-IN') + '</p>';
      html += '<p><strong>Total tax – New regime:</strong> ₹' + newResult.totalTax.toLocaleString('en-IN') + '</p>';

      html += '<p class="highlight">Recommended regime (based only on lower tax): ' + betterRegime + ' regime</p>';
      html += '</div>';

      html += '<div class="result-box">';
      html += '<h3>Tax-saving ideas (Old regime)</h3>';
      html += '<ul>';

      if (remaining80C > 0 && marginalRateOld > 0) {
        html += '<li>You still have unused <strong>80C limit</strong> of about ₹' +
                remaining80C.toLocaleString('en-IN') +
                '. If you choose the Old regime, investing this in 80C options ' +
                '(EPF, PPF, ELSS, home-loan principal, term insurance premium, etc.) ' +
                'could reduce your tax by up to about ₹' +
                potential80CSaving.toLocaleString('en-IN') +
                ' (approx., based on your slab).</li>';
      } else {
        html += '<li>You are already using your full 80C limit of ₹1,50,000, ' +
                'or your taxable income is in a low/zero slab where extra 80C doesn\'t help much.</li>';
      }

      if (remaining80CCD1B > 0 && marginalRateOld > 0) {
        html += '<li>Under <strong>NPS (80CCD(1B))</strong>, you can still invest about ₹' +
                remaining80CCD1B.toLocaleString('en-IN') +
                ' extra and get additional deduction. This could save around ₹' +
                potentialNpsSaving.toLocaleString('en-IN') +
                ' in tax (approx.) in the Old regime.</li>';
      } else {
        html += '<li>You are already at or near the additional NPS limit of ₹50,000 ' +
                'under 80CCD(1B), or your Old-regime slab benefit is low.</li>';
      }

      html += '<li>Premiums for <strong>health insurance</strong> are deductible under 80D ' +
              '(within limits). Beyond tax-saving, adequate health cover ' +
              '(e.g., ₹5–10 lakh family floater) is important for protection.</li>';

      html += '<li>If you have a home loan on a self-occupied house, interest up to ' +
              '₹2,00,000 per year is deductible under <strong>Section 24(b)</strong> ' +
              'in the Old regime.</li>';

      html += '<li>Do not buy poor financial products only to save tax. First check if ' +
              'they suit your goals (returns, lock-in period, risk), then see the tax ' +
              'benefit as a bonus.</li>';

      html += '</ul>';
      html += '</div>';

      resultsDiv.innerHTML = html;
    }
  </script>
